  primary: Color(0xFF1E293B), onPrimary: Colors.white, primaryContainer: Color(0xFFDDE3EA), onPrimaryContainer: Color(0xFF111827),
  surface: Colors.white, onSurface: Color(0xFF111827), onSurfaceVariant: Color(0xFF4B5563),
